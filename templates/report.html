from __future__ import annotations from typing import Any from jinja2 import Environment, FileSystemLoader, select_autoescape def render_report_html(scan_result: dict[str, Any]) -> str: env = Environment( loader=FileSystemLoader("templates"), autoescape=select_autoescape(["html", "xml"]), ) template = env.get_template("report.html") return template.render(result=scan_result)
